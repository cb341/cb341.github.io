Statement succ_eq_add_one n : succ n = n + 1 := by
  rw [one_eq_succ_zero]

Statement zero_le (x : ℕ) : 0 ≤ x := by
  use x
  rw [zero_add]
  rfl

Now our proof:

```lean
Statement le_total (x y : ℕ) : x ≤ y ∨ y ≤ x := by
  induction y with d hd
  right
  exact zero_le x

  cases hd with h1 h2
  left
  cases h1 with e h1
  rw [h1]
  use e + 1


  rw [succ_eq_add_one, add_assoc]
  rfl
  cases h2 with e he
  cases e with a
  rw [he]
  left
  rw [add_zero]
  use 1
  exact succ_eq_add_one d
  right
  use a
  rw [add_succ] at he
  rw [succ_add]
  exact he
```
