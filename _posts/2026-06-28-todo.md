## Final Proof

Axioms
```lean

/- BEGIN PROVE: x ≤ y ∨ y ≤ x -/
induction y with
| zero =>
  right /- 0 ≤ x -/
  exact zero_le x
| succ d hd =>
  /- ind y=0 -/
  /- BEGIN PROVE: x ≤ 0 ∨ 0 ≤ x -/

  /- END  PROVE: x ≤ 0 ∨ 0 ≤ x -/

  /- ind y=d -/
  cases hd with hl hr /- hd = hl ∨ hr
                              |    hr := d ≤ x
                              hl := x ≤ d    
  -/
  cases hl with c hc /- hc := d = x + c -/
  left /- x ≤ succ d -/
  rw[hc]
  use c + 1
  rw[succ_eq_add_one]
  rw[← add_assoc]
  rfl

  /- trial and error -/
  cases hr with e he /- 
  hr : d ≤ x
  he : x = d + e
  -/
  cases e with a /- e -> {0, succ(a)} -/
  /- e=0 ; he : x = d + 0 -/
  rw[add_zero] at he /- he : x = d -/
  left
  rw[he]
  exact le_succ_self d /- le_succ_self x - x ≤ succ x-/
  /- he : x = d + succ a -/
```

```lean
Statement le_total (x y : ℕ) : x ≤ y ∨ y ≤ x := by
  induction y with
  | zero =>
    right
    exact zero_le x
  | succ d hd =>
    cases hd with
    | inl hl =>
      cases hl with c hc
      rw[hc]
      left
      rw[succ_eq_add_one]
      use c + 1
      rw[← add_assoc]
      rfl
    | inr hr =>
      cases hr with c hc
      cases c with
      | zero =>
        rw[zero_eq_0] at hc
        rw[add_zero d] at hc
        left
        rw[hc]
        exact le_succ_self d
      | succ =>
        rw[add_succ] at hc
        right
        rw[hc]
        use a
        rw[succ_add]
        rfl
```
